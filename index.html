# Install dependensi terlebih dahulu:
# pip install yt-dlp fastapi uvicorn
# Pastikan FFmpeg sudah terinstal di sistem Anda

import subprocess
import yt_dlp

def buat_klip_otomatis(url_video: str, mulai_detik: int = 15, durasi: int = 45, output_file: str = "klip_viral_9x16.mp4"):
    """
    Mengunduh video dari YouTube/TikTok/IG lalu memotongnya menjadi
    klip vertikal 9:16 (1080x1920) berdurasi 30-60 detik menggunakan FFmpeg.
    """
    temp_video = "source_temp.mp4"
    
    # 1. Unduh video kualitas terbaik menggunakan yt-dlp
    ydl_opts = {
        'format': 'bestvideo[ext=mp4]+bestaudio[ext=m4a]/best[ext=mp4]',
        'outtmpl': temp_video,
        'quiet': True
    }
    print(f"Mengunduh video dari: {url_video}...")
    with yt_dlp.YoutubeDL(ydl_opts) as ydl:
        ydl.download([url_video])

    # 2. Potong segmen (30s - 60s) & crop tengah ke rasio vertikal 9:16
    print(f"Memotong klip durasi {durasi} detik mulai dari detik ke-{mulai_detik}...")
    ffmpeg_cmd = [
        "ffmpeg", "-y",
        "-ss", str(mulai_detik),
        "-i", temp_video,
        "-t", str(durasi),
        "-vf", "crop=ih*(9/16):ih,scale=1080:1920",
        "-c:v", "libx264", "-preset", "fast", "-crf", "22",
        "-c:a", "aac", "-b:a", "128k",
        output_file
    ]
    subprocess.run(ffmpeg_cmd, check=True)
    print(f"Selesai! File tersimpan di: {output_file}")

if __name__ == "__main__":
    # Contoh pemakaian langsung:
    buat_klip_otomatis("https://www.youtube.com/watch?v=dQw4w9WgXcQ", mulai_detik=20, durasi=45)
